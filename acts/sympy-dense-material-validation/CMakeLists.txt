cmake_minimum_required(VERSION 3.20)
project(DenseAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release)
set(DEPS /Users/andreas/cern/source/acts/ci-dependencies/.spack-env/view)
list(PREPEND CMAKE_PREFIX_PATH ${DEPS})
find_package(Geant4 REQUIRED)
find_package(Eigen3 REQUIRED)
set(ACTS_SOURCE /Users/andreas/cern/source/acts/acts/dev3)
set(ACTS_BUILD /Users/andreas/cern/build/acts/acts/dev3)
set(PATCHED_CODEGEN ${CMAKE_CURRENT_SOURCE_DIR}/patched-codegen CACHE PATH "Generated patched stepper header")
file(GLOB CODEGEN ${ACTS_BUILD}/Core/src/Propagator/codegen/*)
add_executable(audit audit.cpp)
target_include_directories(audit SYSTEM PRIVATE ${DEPS}/include)
target_include_directories(audit PRIVATE ${ACTS_SOURCE}/Core/include ${ACTS_BUILD}/Core/include ${ACTS_SOURCE}/Tests/CommonHelpers/include ${Geant4_INCLUDE_DIRS} ${CODEGEN})
target_link_libraries(audit PRIVATE Eigen3::Eigen ${ACTS_BUILD}/lib/libActsCore.dylib ${Geant4_LIBRARIES})
find_package(Boost REQUIRED COMPONENTS unit_test_framework)
add_library(patched-stepper OBJECT
 ${ACTS_SOURCE}/Core/src/Material/Interactions.cpp
 ${ACTS_SOURCE}/Core/src/Propagator/SympyStepper.cpp
 ${ACTS_SOURCE}/Core/src/Propagator/detail/SympyStepDenseJac.cpp
 ${ACTS_SOURCE}/Core/src/Propagator/detail/SympyStepDenseNoJac.cpp
 ${ACTS_SOURCE}/Core/src/Propagator/detail/SympyStepVacuumJac.cpp
 ${ACTS_SOURCE}/Core/src/Propagator/detail/SympyStepVacuumNoJac.cpp)
target_include_directories(patched-stepper BEFORE PRIVATE ${PATCHED_CODEGEN})
target_include_directories(patched-stepper PRIVATE ${ACTS_SOURCE}/Core/include ${ACTS_BUILD}/Core/include ${CODEGEN})
target_include_directories(patched-stepper SYSTEM PRIVATE ${DEPS}/include)
target_link_libraries(patched-stepper PRIVATE Eigen3::Eigen)
add_executable(audit-patched audit.cpp)
target_link_libraries(audit-patched PRIVATE patched-stepper)
target_include_directories(audit-patched PRIVATE ${ACTS_SOURCE}/Core/include ${ACTS_BUILD}/Core/include ${ACTS_SOURCE}/Tests/CommonHelpers/include ${Geant4_INCLUDE_DIRS})
target_include_directories(audit-patched SYSTEM PRIVATE ${DEPS}/include)
target_link_libraries(audit-patched PRIVATE Eigen3::Eigen ${ACTS_BUILD}/lib/libActsCore.dylib ${Geant4_LIBRARIES})
foreach(kind unit integration material)
 if(kind STREQUAL unit)
  set(test_source ${ACTS_SOURCE}/Tests/UnitTests/Core/Propagator/SympyStepperTests.cpp)
 elseif(kind STREQUAL integration)
  set(test_source ${ACTS_SOURCE}/Tests/IntegrationTests/PropagationSympyDenseConstant.cpp)
 else()
  set(test_source ${ACTS_SOURCE}/Tests/UnitTests/Core/Material/InteractionsTests.cpp)
 endif()
 add_executable(test-${kind} ${test_source})
 target_include_directories(test-${kind} PRIVATE ${ACTS_SOURCE}/Core/include ${ACTS_BUILD}/Core/include ${ACTS_SOURCE}/Tests/CommonHelpers/include ${ACTS_SOURCE}/Examples/Framework/include ${ACTS_SOURCE}/Fatras/include ${ACTS_SOURCE}/Plugins/FpeMonitoring/include)
 target_include_directories(test-${kind} SYSTEM PRIVATE ${DEPS}/include)
 target_compile_definitions(test-${kind} PRIVATE BOOST_TEST_DYN_LINK BOOST_TEST_MODULE=DenseAudit)
 target_link_libraries(test-${kind} PRIVATE patched-stepper Eigen3::Eigen Boost::unit_test_framework ${ACTS_BUILD}/lib/libActsCore.dylib ${ACTS_BUILD}/lib/libActsTestsCommonHelpers.dylib)
endforeach()
